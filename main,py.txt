import os
from dotenv import load_dotenv
from google import genai
import automation
import voice_engine

# Load environment variables
load_dotenv()

raw_key = os.getenv("GEMINI_API_KEY", "")
api_key = raw_key.strip().strip('"').strip("'")

# Initialize Gemini Client
client = genai.Client(api_key=api_key)

SYSTEM_INSTRUCTION = """
Your name is MHZALY. You are a professional AI assistant running locally on a Windows 11 PC.
You speak Urdu, Hindi, English, and Punjabi.
Be polite, respectful, and concise. You can open apps and assist with tasks.
"""

def handle_command(user_text):
    text_lower = user_text.lower()
    
    # Desktop automation
    if "open" in text_lower or "kholo" in text_lower:
        app_name = text_lower.replace("open", "").replace("kholo", "").replace("app", "").strip()
        msg = automation.open_app(app_name)
        voice_engine.speak_text(msg, lang='en')
        return

    # Auto-try available Gemini models
    candidate_models = ['gemini-2.5-flash', 'gemini-2.0-flash-exp', 'gemini-1.5-flash-latest', 'gemini-2.0-flash']
    
    for model_name in candidate_models:
        try:
            response = client.models.generate_content(
                model=model_name,
                contents=user_text,
                config={'system_instruction': SYSTEM_INSTRUCTION}
            )
            if response and response.text:
                voice_engine.speak_text(response.text, lang='ur')
                return
        except Exception as e:
            continue

    print("❌ API connection issue.")
    voice_engine.speak_text("API connection error. Please try again in a moment.", lang='en')

def start_mhzaly():
    print("==========================================")
    print("🚀 MHZALY AI Assistant Active (Windows 11)")
    print("==========================================")
    
    while True:
        user_input = voice_engine.listen_user()
        if user_input:
            if "exit" in user_input.lower() or "band karo" in user_input.lower():
                voice_engine.speak_text("MHZALY offline ho raha hai. Alvida!", lang='ur')
                break
            handle_command(user_input)

if __name__ == "__main__":
    start_mhzaly()
